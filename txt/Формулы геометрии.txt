# Deck: Формулы геометрии
Subject: Математика

// Формулы школьного курса геометрии (без теорем).
// Формат: Название (пояснение: что означают буквы) -> формула.
// Формула сама разбивается на части для сборки: * и пробелы разделяют части,
// ^2 — степень, _0 — индекс. После индекса или степени перед буквой ставится *.

// Треугольник
Периметр треугольника (a, b, c — стороны) -> P = a + b + c
Площадь треугольника (a — сторона, h — высота, проведенная к этой стороне) -> S = ½ah
Площадь прямоугольного треугольника (a, b — катеты) -> S = ½ab
Площадь треугольника через угол (a, b — две стороны, γ — угол между ними) -> S = ½ab sin γ
Полупериметр треугольника (a, b, c — стороны) -> p = (a + b + c)/2
Площадь треугольника по формуле Герона (p — полупериметр, a, b, c — стороны) -> S = √(p(p - a)(p - b)(p - c))
Площадь треугольника через вписанную окружность (p — полупериметр, r — радиус вписанной окружности) -> S = pr
Площадь треугольника через описанную окружность (a, b, c — стороны, R — радиус описанной окружности) -> S = abc/(4R)
Площадь равностороннего треугольника (a — сторона) -> S = a^2√3/4
Высота равностороннего треугольника (a — сторона) -> h = a√3/2
Радиус вписанной окружности равностороннего треугольника (a — сторона) -> r = a√3/6
Радиус описанной окружности равностороннего треугольника (a — сторона) -> R = a√3/3

// Четырехугольники
Периметр прямоугольника (a, b — соседние стороны) -> P = 2(a + b)
Площадь прямоугольника (a, b — соседние стороны) -> S = ab
Диагональ прямоугольника (a, b — соседние стороны) -> d = √(a^2 + b^2)
Площадь квадрата (a — сторона) -> S = a^2
Площадь квадрата через диагональ (d — диагональ) -> S = d^2/2
Диагональ квадрата (a — сторона) -> d = a√2
Площадь параллелограмма (a — сторона, h — высота, проведенная к этой стороне) -> S = ah
Площадь параллелограмма через угол (a, b — соседние стороны, α — угол между ними) -> S = ab sin α
Площадь ромба через диагонали (d_1, d_2 — диагонали) -> S = ½d_1*d_2
Площадь ромба через угол (a — сторона, α — любой угол ромба) -> S = a^2 sin α
Площадь трапеции (a, b — основания, h — высота) -> S = ½(a + b)h
Площадь выпуклого четырехугольника через диагонали (d_1, d_2 — диагонали, α — угол между ними) -> S = ½d_1*d_2 sin α

// Окружность и круг
Длина окружности (r — радиус) -> C = 2πr
Длина окружности через диаметр (d — диаметр) -> C = πd
Площадь круга (r — радиус) -> S = πr^2
Длина дуги окружности (r — радиус, α — центральный угол дуги в градусах) -> l = πrα/180
Площадь кругового сектора (r — радиус, α — центральный угол сектора в градусах) -> S = πr^2*α/360

// Правильный шестиугольник
Площадь правильного шестиугольника (a — сторона) -> S = 3√3a^2/2

// Координаты и векторы
Расстояние между точками (x_1, y_1 и x_2, y_2 — координаты точек) -> d = √((x_2 - x_1)^2 + (y_2 - y_1)^2)
Координата середины отрезка (x_1, x_2 — координаты концов отрезка; y считается так же) -> x = (x_1 + x_2)/2
Длина вектора (x, y — координаты вектора a) -> |a| = √(x^2 + y^2)
Скалярное произведение векторов в координатах (x_1, y_1 и x_2, y_2 — координаты векторов a и b) -> a·b = x_1*x_2 + y_1*y_2
Скалярное произведение векторов через угол (|a|, |b| — длины векторов, α — угол между ними) -> a·b = |a|·|b|·cos α

// Куб и прямоугольный параллелепипед
Объем куба (a — ребро) -> V = a^3
Площадь поверхности куба (a — ребро) -> S = 6a^2
Диагональ куба (a — ребро) -> d = a√3
Объем прямоугольного параллелепипеда (a, b, c — длина, ширина и высота) -> V = abc
Площадь поверхности прямоугольного параллелепипеда (a, b, c — длина, ширина и высота) -> S = 2(ab + bc + ac)
Диагональ прямоугольного параллелепипеда (a, b, c — длина, ширина и высота) -> d = √(a^2 + b^2 + c^2)

// Призма и пирамида
Объем призмы (S_осн — площадь основания, h — высота) -> V = S_осн*h
Площадь боковой поверхности прямой призмы (P_осн — периметр основания, h — высота) -> S_бок = P_осн*h
Объем пирамиды (S_осн — площадь основания, h — высота) -> V = ⅓S_осн*h
Площадь боковой поверхности правильной пирамиды (P_осн — периметр основания, l — апофема, то есть высота боковой грани) -> S_бок = ½P_осн*l

// Тела вращения
Объем цилиндра (r — радиус основания, h — высота) -> V = πr^2*h
Площадь боковой поверхности цилиндра (r — радиус основания, h — высота) -> S_бок = 2πrh
Площадь полной поверхности цилиндра (r — радиус основания, h — высота) -> S = 2πr(r + h)
Объем конуса (r — радиус основания, h — высота) -> V = ⅓πr^2*h
Площадь боковой поверхности конуса (r — радиус основания, l — образующая) -> S_бок = πrl
Площадь полной поверхности конуса (r — радиус основания, l — образующая) -> S = πr(r + l)
Объем шара (R — радиус) -> V = 4/3πR^3
Площадь сферы (R — радиус) -> S = 4πR^2

# Deck: Производные
Subject: Математика
Type: build

// Таблица производных и правила дифференцирования.
// Формат: Название (пояснение) -> формула. Производная слева — одна часть в квадратных скобках:
// [(sin x)′]. Штрих — знак ′. Собирать нужно правую часть.

// Таблица производных
Производная константы (c — постоянная) -> [c′] = 0
Производная степени -> [(x^n)′] = n x^(n-1)
Производная показательной функции (a > 0, a ≠ 1) -> [(a^x)′] = a^x ln a
Производная экспоненты -> [(e^x)′] = e^x
Производная логарифма (a > 0, a ≠ 1) -> [(log_a x)′] = 1/(x ln a)
Производная натурального логарифма -> [(ln x)′] = 1/x
Производная синуса -> [(sin x)′] = cos x
Производная косинуса -> [(cos x)′] = -sin x
Производная корня -> [(√x)′] = 1/(2√x)
Производная тангенса -> [(tg x)′] = 1/cos^2 x
Производная котангенса -> [(ctg x)′] = -1/sin^2 x
Производная арксинуса -> [(arcsin x)′] = 1/√(1 - x^2)
Производная арккосинуса -> [(arccos x)′] = -1/√(1 - x^2)
Производная арктангенса -> [(arctg x)′] = 1/(1 + x^2)
Производная арккотангенса -> [(arcctg x)′] = -1/(1 + x^2)

// Правила дифференцирования (u, v — функции от x, c — постоянная)
Производная суммы -> [(u + v)′] = [u′] + [v′]
Производная разности -> [(u − v)′] = [u′] - [v′]
Вынесение постоянного множителя -> [(cu)′] = c [u′]
Производная произведения -> [(uv)′] = [u′] v + u [v′]
Производная частного -> [(u/v)′] = ([u′] v - u [v′])/v^2
Производная сложной функции -> [(f(g(x)))′] = [f′(g(x))] · [g′(x)]
